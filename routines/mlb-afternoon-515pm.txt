STEP 7 - K PROP MODEL (inline Python - 0 web calls):

python3 << 'PYEOF'
import json, math

LEAGUE_K_PCT = 0.233
CALIBRATION = 0.82

def poisson_p_over(line, mu):
    if mu <= 0: return 0.0
    k = int(line)
    cdf, term = 0.0, math.exp(-mu)
    for i in range(k + 1):
        cdf += term
        if i < k: term *= mu / (i + 1)
    return max(0.0, min(1.0, 1.0 - cdf))

def calibrate(p): return 0.5 + CALIBRATION * (p - 0.5)
def implied(o): return (abs(o)/(abs(o)+100) if o < 0 else 100/(o+100)) if o else None
def tier(e): return ('A' if e>=0.04 else 'B' if e>=0.025 else 'C' if e>=0.01 else 'Pass') if e is not None else 'No Data'

# Fill from Steps 1-5:
candidates = [
    # {'name':'Gerrit Cole','team':'Yankees','opponent':'Blue Jays',
    #  'line':7.5,'over_price':-120,'under_price':-100,
    #  'sp_k_pct':0.29,'opp_k_pct':0.20,'projected_ip':6.0,'umpire_k_adj':0.0},
]

for c in candidates:
    adj = max(0.05, min(0.50, c['sp_k_pct']*(c['opp_k_pct']/LEAGUE_K_PCT)+c.get('umpire_k_adj',0.0)))
    mu = adj * c['projected_ip'] * 4.35
    p_over = calibrate(poisson_p_over(c['line'], mu))
    p_under = 1.0 - p_over
    imp_o, imp_u = implied(c.get('over_price')), implied(c.get('under_price'))
    e_o = round(p_over-imp_o,4) if imp_o else None
    e_u = round(p_under-imp_u,4) if imp_u else None
    print(json.dumps({'player':c['name'],'team':c['team'],'opponent':c['opponent'],
        'line':c['line'],'mean_k':round(mu,2),'p_over':round(p_over,4),'p_under':round(p_under,4),
        'edge_over':e_o,'edge_under':e_u,'tier_over':tier(e_o),'tier_under':tier(e_u),
        'over_price':c.get('over_price'),'under_price':c.get('under_price')}))
PYEOF
